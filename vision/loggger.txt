#!/usr/bin/env python3
import os, json, time, signal, argparse, traceback
from datetime import datetime, timedelta
from pathlib import Path
import numpy as np
from numpy.linalg import norm
from picamera2 import Picamera2
import cv2
from insightface.app import FaceAnalysis

# ---------- CONFIG ----------
DB_DIR   = Path("face_db")
EMB_FILE = DB_DIR / "embeddings.npz"
META_FILE= DB_DIR / "meta.json"

LOG_DIR  = Path("/home/labassist/recognition_logs")
KEEP_DAYS= 3

# ---------- DB HELPERS ----------
def ensure_db():
    DB_DIR.mkdir(parents=True, exist_ok=True)
    if not EMB_FILE.exists():
        # empty DB
        np.savez_compressed(EMB_FILE, embeddings=np.empty((0,512), np.float32), names=np.array([], dtype=object))
    if not META_FILE.exists():
        META_FILE.write_text("{}", encoding="utf-8")

def load_embeddings():
    """Return (E [Nx512], N [N dtype=object], M {name:{role,signature}}). Never crashes."""
    try:
        ensure_db()
        d = np.load(EMB_FILE, allow_pickle=True)
        E = np.array(d["embeddings"], dtype=np.float32)
        N = d["names"]
        try:
            meta = json.loads(META_FILE.read_text(encoding="utf-8"))
        except Exception:
            meta = {}
        if E.ndim == 1:
            E = E.reshape(1, -1)
        if E.size == 0:
            E = np.empty((0,512), np.float32)
            N = np.array([], dtype=object)
        return E, N, meta
    except Exception as e:
        print(f"[WARN] Failed to load embeddings DB: {e}")
        return np.empty((0,512), np.float32), np.array([], dtype=object), {}

# ---------- LOG HELPERS ----------
def ensure_log_dir():
    LOG_DIR.mkdir(parents=True, exist_ok=True)

def today_log_path():
    ensure_log_dir()
    fname = datetime.now().strftime("%Y-%m-%d.jsonl")
    return LOG_DIR / fname

def prune_old_logs(keep_days=KEEP_DAYS):
    cutoff = datetime.now().date() - timedelta(days=keep_days)
    for p in LOG_DIR.glob("*.jsonl"):
        try:
            d = datetime.strptime(p.stem, "%Y-%m-%d").date()
            if d < cutoff:
                p.unlink(missing_ok=True)
        except Exception:
            # ignore files not matching the pattern
            pass

def append_log(obj):
    p = today_log_path()
    with p.open("a", encoding="utf-8") as f:
        f.write(json.dumps(obj, ensure_ascii=False) + "\n")

# ---------- UTILS ----------
def clean_ascii(s):
    if s is None: return ""
    if not isinstance(s, str): s = str(s)
    return s.encode("ascii", "ignore").decode().replace("?", "").strip()

def label_for(name, role, score, sig):
    name = clean_ascii(name); role = clean_ascii(role); sig = clean_ascii(sig)
    role_txt = f" ({role})" if role else ""
    sig_txt  = f" Sig:{sig}" if sig else ""
    return f"Verified: {name}{role_txt} {score:.2f}{sig_txt}".strip()

# ---------- RECOGNITION CORE ----------
def recognize_frame(app, frame, E, N, M, sim_thresh):
    """
    Returns:
      faces_out: list of dict {name/UNKNOWN, role, score, bbox}
      counts: dict with student/staff/unknown/total
    """
    faces_out = []
    counts = {"student": 0, "staff":  0, "unknown": 0, "total": 0}
    faces = app.get(frame)
    h, w = frame.shape[:2]

    for f in faces:
        x1,y1,x2,y2 = map(int, f.bbox)
        x1,y1 = max(0,x1), max(0,y1)
        x2,y2 = min(w-1,x2), min(h-1,y2)
        name = "UNKNOWN"; role = ""; score = None

        fe = getattr(f, "normed_embedding", None)
        if E.size and fe is not None and fe.size == 512:
            emb = fe.astype(np.float32)
            emb = emb / (norm(emb) + 1e-9)
            sims = E @ emb / (np.linalg.norm(E, axis=1) + 1e-9)
            idx  = int(np.argmax(sims)); sc = float(sims[idx])
            if sc >= sim_thresh:
                nm = N[idx]
                name = clean_ascii(nm)
                meta_entry = M.get(str(nm), {})
                role = meta_entry.get("role", "")
                score = sc

        if name == "UNKNOWN":
            counts["unknown"] += 1
        elif role == "student":
            counts["student"] += 1
        elif role == "staff":
            counts["staff"] += 1
        else:
            # recognized but role missing -> treat as unknown for counts
            counts["unknown"] += 1

        faces_out.append({
            "name": name,
            "role": role,
            "score": score,
            "bbox": [x1,y1,x2,y2],
        })

    counts["total"] = len(faces_out)
    return faces_out, counts

# ---------- MAIN LOOP ----------
stop_flag = False
def handle_sigint(sig, frame):
    global stop_flag
    stop_flag = True
signal.signal(signal.SIGINT, handle_sigint)
signal.signal(signal.SIGTERM, handle_sigint)

def main():
    parser = argparse.ArgumentParser(description="Headless face recognition logger (Pi, no streaming)")
    parser.add_argument("--fmt", choices=["yuv","rgb","bgr"], default="yuv", help="camera output pixel format")
    parser.add_argument("--width", type=int, default=640)
    parser.add_argument("--height", type=int, default=480)
    parser.add_argument("--det-interval", type=int, default=6, help="detect every N frames")
    parser.add_argument("--det-width", type=int, default=320, help="InsightFace det_size (square)")
    parser.add_argument("--thresh", type=float, default=0.35, help="cosine similarity threshold")
    parser.add_argument("--keep-days", type=int, default=KEEP_DAYS, help="days to keep daily .jsonl logs")
    args = parser.parse_args()

    ensure_db()
    ensure_log_dir()
    prune_old_logs(args.keep_days)

    # Load DB once (you can re-load periodically if needed)
    E, N, M = load_embeddings()

    # InsightFace (CPU)
    app = FaceAnalysis(providers=['CPUExecutionProvider'])
    app.prepare(ctx_id=0, det_size=(args.det_width, args.det_width))
    print(f"set det-size: ({args.det_width}, {args.det_width})")

    # Camera (Picamera2) — no V4L2 timeouts
    cam = Picamera2()
    fmt = args.fmt.lower()
    if fmt == "yuv":
        main_cfg = {"size": (args.width, args.height), "format": "YUV420"}  # fast, accurate colors
        convert_mode = "yuv2bgr"
    elif fmt == "rgb":
        main_cfg = {"size": (args.width, args.height), "format": "RGB888"}
        convert_mode = "rgb2bgr"
    else:
        main_cfg = {"size": (args.width, args.height), "format": "BGR888"}
        convert_mode = "bgr"

    cfg = cam.create_video_configuration(main=main_cfg, buffer_count=4)
    cam.configure(cfg)
    cam.start()
    time.sleep(1.0)  # AE/AWB settle

    # State for change-logging
    prev_names = set()     # recognized names currently in frame
    prev_counts = {"student":0,"staff":0,"unknown":0,"total":0}
    frame_idx = 0

    print(f"Logging to: {today_log_path()}")
    append_log({"timestamp": datetime.now().isoformat(), "event":"logger_started"})

    try:
        while not stop_flag:
            # --- capture frame ---
            if convert_mode == "yuv2bgr":
                yuv = cam.capture_array("main")           # I420
                frame = cv2.cvtColor(yuv, cv2.COLOR_YUV2BGR_I420)
            elif convert_mode == "rgb2bgr":
                rgb = cam.capture_array("main")
                frame = cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)
            else:
                frame = cam.capture_array("main")         # BGR888 already

            # --- detect every N frames ---
            if frame_idx % max(1, args.det_interval) == 0:
                faces_out, counts = recognize_frame(app, frame, E, N, M, args.thresh)

                # current set of recognized names (exclude UNKNOWN)
                curr_names = set([f["name"] for f in faces_out if f["name"] != "UNKNOWN"])

                # fire events on enter/leave for recognized names
                entered = sorted(list(curr_names - prev_names))
                left    = sorted(list(prev_names - curr_names))

                now_iso = datetime.now().isoformat()

                for nm in entered:
                    append_log({"timestamp": now_iso, "event": "enter", "name": nm})

                for nm in left:
                    append_log({"timestamp": now_iso, "event": "leave", "name": nm})

                # fire a counts_change event only if counts changed
                if counts != prev_counts:
                    append_log({
                        "timestamp": now_iso,
                        "event": "counts_change",
                        "counts": counts,
                        "recognized": sorted(list(curr_names))
                    })

                prev_names = curr_names
                prev_counts = counts

            frame_idx += 1

            # small sleep to keep CPU sane; camera is ~30fps
            time.sleep(0.01)

    except Exception as e:
        print("[ERROR] Unhandled exception in main loop:")
        traceback.print_exc()
        append_log({"timestamp": datetime.now().isoformat(), "event":"error", "message": str(e)})
    finally:
        try:
            cam.stop()
        except Exception:
            pass
        append_log({"timestamp": datetime.now().isoformat(), "event":"logger_stopped"})
        # prune once more at shutdown
        prune_old_logs(args.keep_days)
        print("Stopped.")

if __name__ == "__main__":
    main()
